| telemetry.py | TelemetryIngest, TelemetryResponse |
| alert.py | AlertRuleCreate, AlertRuleUpdate, AlertRuleResponse, AlertResponse |
| ticket.py | TicketCreate, TicketUpdate, TicketResponse |
| user.py | UserRegister, UserLogin, UserResponse, TokenResponse |
